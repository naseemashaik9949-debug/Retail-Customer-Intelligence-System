CREATE OR REPLACE VIEW CUSTOMER_PROFITABILITY_VIEW AS
SELECT CUSTOMER_ID,
       CUSTOMER_NAME,
       MAX(SEGMENT) AS SEGMENT,
       ROUND(SUM(SALES), 2) AS TOTAL_SALES,
       ROUND(SUM(PROFIT), 2) AS TOTAL_PROFIT
FROM SUPERSTORE_RAW
GROUP BY CUSTOMER_ID,
         CUSTOMER_NAME;
Step2:
         
SELECT *
FROM CUSTOMER_PROFITABILITY_VIEW;
